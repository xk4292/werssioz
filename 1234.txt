DROP TABLE IF EXISTS water_measurements CASCADE;
DROP TABLE IF EXISTS pipeline_sections CASCADE;
DROP TABLE IF EXISTS consumption_districts CASCADE;
DROP TABLE IF EXISTS pipeline_types CASCADE;

CREATE TABLE pipeline_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    diameter INTEGER NOT NULL,
    material VARCHAR(100) NOT NULL,
    pressure_standard DECIMAL(10, 2) NOT NULL,
    service_life INTEGER NOT NULL,
    UNIQUE(name, diameter)
);

CREATE TABLE pipeline_sections (
    id SERIAL PRIMARY KEY,
    pipeline_type_id INTEGER NOT NULL REFERENCES pipeline_types(id) ON DELETE RESTRICT,
    section_number VARCHAR(50) NOT NULL UNIQUE,
    status VARCHAR(50) NOT NULL,
    installation_date DATE NOT NULL,
    length DECIMAL(10, 2) NOT NULL
);

CREATE TABLE consumption_districts (
    id SERIAL PRIMARY KEY,
    district_name VARCHAR(100) NOT NULL UNIQUE,
    house_count INTEGER NOT NULL,
    dispatch_address TEXT NOT NULL,
    phone VARCHAR(20) NOT NULL
);

CREATE TABLE water_measurements (
    id SERIAL PRIMARY KEY,
    pipeline_section_id INTEGER NOT NULL REFERENCES pipeline_sections(id) ON DELETE CASCADE,
    consumption_district_id INTEGER NOT NULL REFERENCES consumption_districts(id) ON DELETE CASCADE,
    measurement_time TIMESTAMP NOT NULL,
    pressure DECIMAL(10, 2) NOT NULL,
    flow_volume DECIMAL(10, 2) NOT NULL
);

CREATE INDEX idx_measurements_time ON water_measurements(measurement_time);
